## 3. Remediation for Issue 3: Technical Artifacts Leaking into Final Text

### 3.1 Objective
Prevent markdown headers (`# Chapter`, `## Title`), conversational preambles (`Ось фінальний переклад:`), and unmapped `<tag_1>` tags from leaking into final documents.

### 3.2 Dynamic Multi-Pass Regex Sanitizer
Replace static `.startswith()` checks in `src/translation/aya_editing_engine.py:395-487` (`_sanitize_output`) with comprehensive regex pattern matching:

```python
import re

def _sanitize_output(self, text: str) -> str:
    if not text:
        return ""
    
    cleaned = text.strip()
    
    for _ in range(5):
        prev = cleaned
        
        # 1. Strip markdown code fences (```markdown ... ``` or ``` ...)
        cleaned = re.sub(r"^```(?:[a-zA-Z0-9_-]+)?\s*\n?", "", cleaned, flags=re.IGNORECASE)
        cleaned = re.sub(r"\n?```\s*$", "", cleaned)
        
        # 2. Strip echoed prompt section headers (e.g., === ВІДРЕДАГОВАНИЙ ТЕКСТ ===)
        cleaned = re.sub(r"^={3,}\s*[^\n]+?\s*={3,}\s*\n?", "", cleaned, flags=re.MULTILINE)
        
        # 3. Strip markdown headings generated by LLM (e.g., # Глава 1: ..., ## Розділ ...)
        cleaned = re.sub(r"^#{1,6}\s+[^\n]+\n*", "", cleaned, flags=re.MULTILINE)
        
        # 4. Strip conversational preambles
        cleaned = re.sub(
            r"^(?:\*{1,2})?(?:Ось\s+(?:фінальний|виправлений|відредагований|готовий)\s+переклад|Відредагований\s+текст|Фінальний\s+переклад|Переклад|Ukrainian\s+translation)(?:\*{1,2})?\s*:\s*\n*",
            "",
            cleaned,
            flags=re.IGNORECASE
        )
        
        # 5. Strip unmapped inline tag artifacts (e.g., <tag_1>, </tag_1>, <tag_12>)
        cleaned = re.sub(r"</?tag_\d+>", "", cleaned)
        
        # 6. Strip outer wrapping quotes if entire text is enclosed
        if (cleaned.startswith('"') and cleaned.endswith('"')) or \
           (cleaned.startswith('«') and cleaned.endswith('»')) or \
           (cleaned.startswith('“') and cleaned.endswith('”')):
            if cleaned.count('"') == 2 or cleaned.count('«') == 1:
                cleaned = cleaned[1:-1].strip()
        
        cleaned = cleaned.strip()
        if cleaned == prev:
            break
            
    return cleaned
```

       matching_terms = []

def sanitize_mixed_script_words(text: str) -> str:
    # Matches words containing both Cyrillic and Latin characters
    mixed_word_pattern = re.compile(
        r'\b(?=[а-яіїєґА-ЯІЇЄҐ]*[a-zA-Z])(?=[a-zA-Z]*[а-яіїєґА-ЯІЇЄҐ])[а-яіїєґА-ЯІЇЄҐa-zA-Z]+\b'
    )
    # Known portmanteau fixup map or transliteration fallback
    def fix_mixed(match):
        word = match.group(0)
        if "Смачнissimo" in word or "смачнissimo" in word:
            return "Смакота" if word[0].isupper() else "смакота"
        # Strip trailing Latin suffix if attached to Cyrillic stem
        return re.sub(r'[a-zA-Z]+$', '', word)
        
    return mixed_word_pattern.sub(fix_mixed, text)
```

# src/preprocessing/unit_converter.py
import re
from typing import Tuple

class UnitConverter:
    """
    Deterministic rule-based unit converter for English to Ukrainian literary translation.
    """
    CONVERSION_FACTORS = {
        'foot': (0.3048, 'метр', 'метри', 'метрів'),
        'feet': (0.3048, 'метр', 'метри', 'метрів'),
        'inch': (2.54, 'сантиметр', 'сантиметри', 'сантиметрів'),
        'inches': (2.54, 'сантиметр', 'сантиметри', 'сантиметрів'),
        'yard': (0.9144, 'метр', 'метри', 'метрів'),
        'yards': (0.9144, 'метр', 'метри', 'метрів'),
        'mile': (1.60934, 'кілометр', 'кілометри', 'кілометрів'),
        'miles': (1.60934, 'кілометр', 'кілометри', 'кілометрів'),
        'pound': (0.453592, 'кілограм', 'кілограми', 'кілограмів'),
        'pounds': (0.453592, 'кілограм', 'кілограми', 'кілограмів'),
    }

    WORD_NUMBERS = {
        'one': 1, 'two': 2, 'three': 3, 'four': 4, 'five': 5,
        'six': 6, 'seven': 7, 'eight': 8, 'nine': 9, 'ten': 10,
        'fifteen': 15, 'twenty': 20, 'thirty': 30, 'forty': 40,
        'fifty': 50, 'sixty': 60, 'seventy': 70, 'eighty': 80,
        'ninety': 90, 'hundred': 100
    }

    @classmethod
    def convert_feet_to_meters(cls, value: float) -> Tuple[float, str]:
        meters = value * 0.3048
        if meters >= 10:
            rounded = round(meters)
        else:
            rounded = round(meters, 1)
        return rounded, cls.get_ukrainian_unit(rounded, ('метр', 'метри', 'метрів'))

    @staticmethod
    def get_ukrainian_unit(num: float, forms: Tuple[str, str, str]) -> str:
        int_part = int(round(num))
        if 11 <= (int_part % 100) <= 19:
            return forms[2]
        remainder = int_part % 10
        if remainder == 1:
            return forms[0]
        elif 2 <= remainder <= 4:
            return forms[1]
        return forms[2]
```
